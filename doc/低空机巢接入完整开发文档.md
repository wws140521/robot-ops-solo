## 第一章 定位与边界

### 1.1 一句话定位

> Robot-Ops-Solo 是**跨品牌异构智能设备运维中台**。首期覆盖地面工业机器人（FANUC / KUKA / 埃斯顿），二期将无人机自动机巢、eVTOL 起降场边缘设备作为 `device_class` 的一类，接入同一套 UDM、Adapter 工厂、SOP 引擎、AI 健康分与轻量 OTA。

### 1.2 边界矩阵（必须遵守）

| 低空侧对象 | 是否纳入 | 说明 |
|-----------|:-------:|------|
| 无人机自动机巢（大疆机场 DJI Dock 2/3、科比特、普宙等） | ✅ | 本质 = 边缘网关 + 执行机构，与机器人边缘盒子**同构** |
| 机巢充电模块、舱门、升降平台 | ✅ | 温度/电压/在线率/故障码，与机器人关节监控**同逻辑** |
| 机巢气象站（风速/雨量/温湿度） | ✅ | 映射为 `sensor` 型遥测 |
| 5G-A 通感基站边缘节点 | ✅ | 边缘节点运维，映射为 `gateway` 型设备 |
| eVTOL 起降场地面设施（充电接口、消防、照明） | ✅ | 地面基建，不涉及飞行器本体 |
| **无人机本体飞控/航线/载荷任务** | ❌ | 留给无人机中台厂商（大疆、纵横、科比特等） |
| **空域申请 / UOM 实名 / 飞行计划** | ❌ | 空管平台职责，不碰 |
| **eVTOL 适航 / 飞控固件 OTA** | ❌ | 民航适航管辖（2026 新《民用航空法》专章），需持证 |
| **飞行调度 / 冲突预警** | ❌ | 飞行运营平台职责 |

### 1.3 设计原则

1. **UDM 单向扩展**：新增 `device_class` 枚举 + 可选扩展字段，**不破坏**现有地面机器人模型
2. **Adapter 同构**：新建设备适配器，复用 `adaptByBrand()` 工厂与 `AdaptedResult` 输出契约
3. **能力对齐**：机巢告警 → SOP 步骤 → Agent 解释 → OTA 升级机巢边缘节点，**完整复用**
4. **数据不出境、不涉飞行**：采集数据仅为设备健康与运行状态，**不含飞行轨迹、任务载荷**

```typescript
// packages/adapter-kit/src/adapters/registry.ts

import { adaptDJIDock } from './dji-dock';
import { adaptAutelDock } from './autel-dock';
import { adaptVertiport } from './vertiport';

export const BRAND_REGISTRY: Record<string, Adapter> = {
  // —— 原有 ——
  fanuc: { adapt: adaptFANUC, device_class: 'ground_robot' },
  kuka: { adapt: adaptKUKA, device_class: 'ground_robot' },
  estun: { adapt: adaptEstun, device_class: 'ground_robot' },

  // —— 新增：低空 ——
  'dji-dock': { adapt: adaptDJIDock, device_class: 'uav_dock' },
  'autel-dock': { adapt: adaptAutelDock, device_class: 'uav_dock' },
  'generic-vertiport': { adapt: adaptVertiport, device_class: 'vertiport' },
};

export function adaptByBrand(brand: string, raw: any): AdaptedResult {
  const entry = BRAND_REGISTRY[brand.toLowerCase()];
  if (!entry) throw new Error(`Unsupported brand: ${brand}`);
  return entry.adapt(raw);
}
```

### 3.2 DJI Dock 2/3 适配器

```typescript
// packages/adapter-kit/src/adapters/dji-dock.ts
import {
  UnifiedRobotState, AdaptedResult, UAV_ALARM_CODES,
} from '../types/unified';

/**
 * 输入：大疆机场开放 API 返回的 JSON（经边缘侧 fetch 或 webhook）
 * 输出：统一模型
 */
export function adaptDJIDock(raw: any): AdaptedResult {
  const dock = raw.dock ?? raw;
  const uav = raw.uav ?? null;

  const alarms: UnifiedAlarm[] = [];
  if (dock.charger_temperature_c > 60) {
    alarms.push({ code: UAV_ALARM_CODES.DOCK_CHARGER_OVER_TEMP, level: 'major', ts: Date.now() });
  }
  if (dock.door_state === 'jammed') {
    alarms.push({ code: UAV_ALARM_CODES.DOCK_DOOR_JAMMED, level: 'critical', ts: Date.now() });
  }
  if (uav?.battery_percent < 15) {
    alarms.push({ code: UAV_ALARM_CODES.UAV_BATTERY_LOW, level: 'major', ts: Date.now() });
  }
  if ((uav?.motor_temperatures ?? []).some((t: number) => t > 75)) {
    alarms.push({ code: UAV_ALARM_CODES.UAV_MOTOR_OVER_TEMP, level: 'major', ts: Date.now() });
  }

  const state: UnifiedRobotState = {
    robot_id: `DJI_DOCK_${raw.sn ?? raw.dock_sn}`,
    brand: 'dji-dock',
    model: raw.product_version ?? 'Dock 2',
    status: alarms.some(a => a.level === 'critical') ? 'alarm'
      : dock.dock_state === 'maintenance' ? 'maintenance' : 'online',
    joints: [],  // 机巢无关节，留空（UDM 兼容）
    alarms,
    runtime: { power_on_hours: dock.flight_total_hours ?? 0, last_seen: new Date().toISOString() },
    health_score: 0,  // 由健康分引擎填充
    device_class: 'uav_dock',
    dock: {
      dock_state: dock.dock_state ?? 'idle',
      charger_temp_c: dock.charger_temperature_c ?? 0,
      charger_voltage_v: dock.charger_voltage_v ?? 0,
      charger_current_a: dock.charger_current_a ?? 0,
      door_state: dock.door_state ?? 'closed',
      lift_platform: dock.lift_platform_state ?? 'down',
      weather: {
        wind_speed_mps: dock.wind_speed ?? 0,
        wind_gust_mps: dock.wind_gust ?? 0,
        rainfall_mm: dock.rainfall ?? 0,
        temperature_c: dock.temperature ?? 0,
        humidity_pct: dock.humidity ?? 0,
      },
      has_uav_inside: dock.uav_inside ?? false,
    },
    uav: uav ? {
      battery_pct: uav.battery_percent ?? 0,
      battery_cycles: uav.battery_cycles ?? 0,
      signal_rssi: uav.signal_rssi ?? -90,
      gps_satellites: uav.gps_satellites ?? 0,
      motor_temps: uav.motor_temperatures ?? [],
      propeller_rpm: uav.propeller_rpms ?? [],
      last_flight_id: uav.last_flight_id,
    } : undefined,
  };

  return { state, warnings: [] };
}
```

```python
# roboticsops-edge/collectors/dji_dock_collector.py
"""
DJI Dock 采集器：通过大疆开放 API（HTTP / WebSocket）拉取机巢与无人机状态，
归一化为 adapter-kit 可消费的 raw dict，再由边缘侧转换后发布 MQTT。
"""
import time
import requests
from typing import Optional


class DJIDockCollector:
    def __init__(self, dock_sn: str, api_base: str, app_key: str, app_secret: str):
        self.dock_sn = dock_sn
        self.api_base = api_base.rstrip('/')
        self.app_key = app_key
        self.app_secret = app_secret
        self._token: Optional[str] = None
        self._token_expire = 0

    # —— 鉴权（示意，实际按大疆开放平台文档）——
    def _get_token(self) -> str:
        if self._token and time.time() < self._token_expire - 60:
            return self._token
        resp = requests.post(f"{self.api_base}/oauth/token", json={
            "app_key": self.app_key, "app_secret": self.app_secret,
        }, timeout=5)
        resp.raise_for_status()
        data = resp.json()
        self._token = data["access_token"]
        self._token_expire = time.time() + data.get("expires_in", 7200)
        return self._token

    def _request(self, path: str) -> dict:
        headers = {"Authorization": f"Bearer {self._get_token()}"}
        resp = requests.get(f"{self.api_base}{path}", headers=headers, timeout=5)
        resp.raise_for_status()
        return resp.json()

    def collect(self) -> dict:
        """归一化为 adapter-kit dji-dock 输入的 raw 结构"""
        dock = self._request(f"/v1/docks/{self.dock_sn}/state")
        uav = None
        try:
            uav = self._request(f"/v1/docks/{self.dock_sn}/uav")
        except Exception:
            pass  # 机巢内无无人机时忽略

        return {
            "sn": self.dock_sn,
            "product_version": dock.get("product_version"),
            "dock_state": dock.get("state"),
            "charger_temperature_c": dock.get("charger", {}).get("temperature_c", 0),
            "charger_voltage_v": dock.get("charger", {}).get("voltage_v", 0),
            "charger_current_a": dock.get("charger", {}).get("current_a", 0),
            "door_state": dock.get("door", {}).get("state", "closed"),
            "door_jammed": dock.get("door", {}).get("jammed", False),
            "lift_platform_state": dock.get("lift", {}).get("state", "down"),
            "wind_speed": dock.get("weather", {}).get("wind_speed_mps", 0),
            "wind_gust": dock.get("weather", {}).get("wind_gust_mps", 0),
            "rainfall": dock.get("weather", {}).get("rainfall_mm", 0),
            "temperature": dock.get("weather", {}).get("temperature_c", 0),
            "humidity": dock.get("weather", {}).get("humidity_pct", 0),
            "uav_inside": dock.get("uav_inside", False),
            "uav": uav and {
                "battery_percent": uav.get("battery", {}).get("percent", 0),
                "battery_cycles": uav.get("battery", {}).get("cycles", 0),
                "signal_rssi": uav.get("link", {}).get("rssi", -90),
                "gps_satellites": uav.get("gps", {}).get("satellites", 0),
                "motor_temperatures": uav.get("motors", []),
                "propeller_rpms": uav.get("propellers", []),
                "last_flight_id": uav.get("last_flight_id"),
            } or None,
        }


if __name__ == "__main__":
```

## 第六章 健康分算法扩展

> 复用 `ai/health-score.ts` 的加权框架，为每类设备定义**独立权重**，结果归一化到 0-100。

### 6.1 机巢健康分

```typescript
// packages/ai/src/health-score.ts（扩展）

export interface HealthWeights {
  chargerTemp: number;
  doorState: number;
  batteryHealth: number;
  signal: number;
  weather: number;
}

export const DOCK_WEIGHTS: HealthWeights = {
  chargerTemp: 0.30,    // 充电器温度（最高权重）
  doorState: 0.20,      // 舱门状态
  batteryHealth: 0.25,   // 电池健康（循环次数 + 电量）
  signal: 0.10,         // 图传信号
  weather: 0.15,        // 气象（风速/雨量）
};

/**
 * 计算机巢健康分（0-100，越高越健康）
 */
export function calcDockHealthScore(
  dock: DockTelemetry,
  uav: UAVTelemetry | undefined,
): number {
  const scores: Record<keyof HealthWeights, number> = {
    // 充电器温度：>60℃ 为 0，<40℃ 为 100
    chargerTemp: clamp(100 - Math.max(0, dock.charger_temp_c - 40) * 5, 0, 100),
    // 舱门：jammed=0, 其余按状态
    doorState: dock.door_state === 'jammed' ? 0
      : dock.door_state === 'open' ? 80 : 100,
    // 电池健康：循环次数（>300 衰减）+ 当前电量
    batteryHealth: uav ? clamp(100 - Math.max(0, uav.battery_cycles - 100) * 0.5, 0, 100) * 0.6
      + uav.battery_pct * 0.4 : 100,
    // 信号：RSSI -90=0, -50=100
    signal: uav ? clamp((uav.signal_rssi + 90) * 2.5, 0, 100) : 100,
    // 气象：风速>8m/s 开始衰减
    weather: clamp(100 - Math.max(0, dock.weather.wind_speed_mps - 8) * 12, 0, 100),
  };

  const { chargerTemp, doorState, batteryHealth, signal, weather } = scores;
  const w = DOCK_WEIGHTS;
  return Math.round(
    chargerTemp * w.chargerTemp
    + doorState * w.doorState
    + batteryHealth * w.batteryHealth
    + signal * w.signal
    + weather * w.weather,
  );
}

function clamp(v: number, min: number, max: number): number {
  return Math.min(Math.max(v, min), max);
}
```

### 6.2 统一入口（路由到对应算法）

```typescript

import { calcDockHealthScore } from './dock-health';
import { calcVertiportHealthScore } from './vertiport-health';
// import { calcRobotHealthScore } from './robot-health';  // 原有

export function calcHealthScore(state: UnifiedRobotState): number {
  switch (state.device_class) {
    case 'uav_dock':
      return calcDockHealthScore(state.dock!, state.uav);
    case 'vertiport':
      return calcVertiportHealthScore(state.vertiport!);

}
```

## 第十三章 合规边界（低空专项）

### 13.1 红线（一票否决）

1. **绝不升级飞控/适航软件**：无人机飞控、eVTOL 航电固件 OTA 一律不做
2. **绝不接入飞行管控**：不碰空域申请、飞行计划、UOM 实名、冲突预警
3. **绝不采集飞行轨迹/载荷**：采集数据仅为设备健康与运行状态
4. **数据不出境**：机巢数据同机器人数据，遵循客户数据不出厂约定
5. **机巢操作安全**：远程开舱门/复位须二次确认 + 作业时段窗口限制

### 13.2 客户自部署主体归属（复用合规逻辑）

- 机巢部署在**客户云账号 + 客户备案域名** → 信息服务主体为客户，你收技术服务费
- 合同定性：**"异构智能设备（含无人机机巢）运维系统集成与技术支持"**
- 不办 ICP 证（B2B 项目制），只做 ICP 备案（你官网）

### 13.3 与民航法规的关系

- 2026 新《民用航空法》对 eVTOL/无人机适航有专章规定
- 你的定位是**地面设备运维**，不涉及适航审定，无需民航局运行合格证
- 若客户要求你"代其申请飞行资质"，明确拒绝或转介专业机构
